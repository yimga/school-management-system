Tenant page blank / stale UI — cache diagnostic

Do not open this file for the test. Log into your tenant site (e.g. new-school.runmycampus.com), open DevTools → Console, paste the script below, press Enter. Copy the printed object back here if it still looks wrong.

Server check (2026-06-21): Live host already serves SW v4.04.42 and rmc-wizard.css with min-height: auto. If the page still looks broken, the problem is almost always your browser still running an old service worker or Cache Storage bucket.

Why caches.keys() showed nothing

It returns a Promise. Use await caches.keys() or the script below — not bare caches.keys().

Paste this on the tenant origin

(async function rmcCacheDiagnostic() {
  const out = { ts: new Date().toISOString(), url: location.href };
  out.secureContext = window.isSecureContext;
  out.swApi = "serviceWorker" in navigator;
  out.cachesApi = "caches" in window;
  try {
    out.cacheKeys = await caches.keys();
  } catch (e) {
    out.cacheKeysError = String(e);
  }
  const reg = await navigator.serviceWorker.getRegistration();
  out.swRegistered = !!reg;
  out.pageControlledBySw = !!navigator.serviceWorker.controller;
  out.activeUrl = reg?.active?.scriptURL || null;
  out.activeState = reg?.active?.state || null;
  out.hasWaitingWorker = !!reg?.waiting;
  out.hasInstallingWorker = !!reg?.installing;
  out.pwaConfig = window.SMS_OFFLINE_CONFIG ? {
    pwaEnabled: window.SMS_OFFLINE_CONFIG.pwaEnabled,
    enabled: window.SMS_OFFLINE_CONFIG.enabled,
  } : null;
  if (reg?.active?.scriptURL) {
    try {
      const swText = await (await fetch(reg.active.scriptURL, { cache: "no-store" })).text();
      out.activeSwCacheVersion = (swText.match(/const CACHE_VERSION = "([^"]+)"/) || [])[1] || null;
    } catch (e) {
      out.activeSwFetchError = String(e);
    }
  }
  try {
    const css = await (await fetch("/static/css/rmc-wizard.css", { cache: "reload" })).text();
    const block = (css.match(/\.rmc-wizard-shell\s*\{[\s\S]*?\}/) || [""])[0];
    out.wizardShellHasAutoMinHeight = block.includes("min-height: auto");
    out.wizardShellSnippet = block.slice(0, 120);
  } catch (e) {
    out.cssFetchError = String(e);
  }
  const shell = document.querySelector(".rmc-wizard-shell");
  if (shell) {
    const cs = getComputedStyle(shell);
    out.liveShellMinHeight = cs.minHeight;
    out.reviewMode = !document.body.classList.contains("rmc-wizard-shell-viewport-locked");
  }
  out.revealArmed = document.documentElement.getAttribute("data-rmc-reveal-armed");
  out.hiddenRevealCount = document.querySelectorAll(".rmc-reveal:not(.is-revealed)").length;
  console.log("RMC cache diagnostic:", out);
  return out;
})();

How to read results

  • activeSwCacheVersion should be sms-v4.04.42-wizard-review-void-fix-2026-06-20
  • cacheKeys should list names containing v4.04.42 (not only v4.04.39 or older)
  • hasWaitingWorker: true → click “Refresh” on the update toast or unregister SW
  • pageControlledBySw: false → hard refresh once; if still false, SW may be blocked
  • wizardShellHasAutoMinHeight: false → browser still serving old CSS from cache

Hard reset (if diagnostic shows old version)

  1. DevTools → Application → Service Workers → Unregister
  2. Application → Storage → Clear site data
  3. Close all tabs for that hostname
  4. Reopen → hard reload (Ctrl+Shift+R)
  5. Run the diagnostic script again
