# Optional dependencies — opt-in features. Not required to run the platform.

# Install all of them:

# Wave 10: true per-row Shapley values for at-risk model explanations.
# Without this, `explain_score(method="shap")` soft-falls-through to the
# fast (feature_importances_) path. ~50 MB install footprint.
shap>=0.44.0

# Wave 9: pgvector Python adapter — only needed if you also enable the
# Postgres extension (operator-installed) and flip PGVECTOR_ENABLED=True.
# The platform uses raw SQL for cosine ranking, so the Python package is
# strictly optional for the ORM-side migration helper.
# pgvector>=0.2.5

# Object storage (S3-compatible) for durable, scalable media — only needed when
# MEDIA_STORAGE_BACKEND=s3 (or AWS_S3_ENDPOINT_URL + AWS_STORAGE_BUCKET_NAME) is
# set. Works with self-hosted MinIO (FOSS), Cloudflare R2, Backblaze B2, AWS S3,
# or a regional/edge bucket. The default local-filesystem media path needs none
# of this. See config/settings.py STORAGES block.
django-storages[s3]>=1.14  # pulls boto3

# boto3 is ALSO the durable secondary for tenant DR immutable snapshots
# (apps/lifecycle/tenant_dr_snapshot.py). When TENANT_SNAPSHOT_S3_BUCKET is set,
# each daily snapshot is uploaded to object storage — the only store that
# survives an ephemeral-container redeploy (local var/ is wiped). django-storages
# above already pulls boto3; this explicit pin documents the DR dependency and
# lets a deploy that wants ONLY the DR S3 leg (not S3 media) install boto3 alone.
# NOTE: build.sh installs requirements.txt only — an operator enabling S3 DR must
# also install this file (or add boto3 to the build) on the WORKER service, which
# is where the capture task runs. See docs/DR_SELF_HOST_RESTORE_RUNBOOK.md §2.
boto3>=1.34

# Academics CP-SAT timetable solver (metric 15). Intentionally NOT in

# ortools activates ``apps.academics.scheduling_solver._solve_with_ortools`` for
# operators who want the optional CP-SAT backend (see apps/academics/README.md).
# ~100 MB install footprint; binary wheels for CPython 3.10–3.12.
ortools>=9.10.4067
